```python
import json

def get_max_len(data_list, field):
    """Dynamically find the longest string to set the PIC X sizes."""
    lengths = [len(str(item.get(field, ""))) for item in data_list]
    return max(lengths) if lengths else 20

def generate_cobol_copybook(json_file, output_file):
    with open(json_file, 'r') as f:
        data = json.load(f)

    place_structures = data.get('place_structures', [])
    evidentials = data.get('evidentials', [])

    # Calculate dynamic lengths for the text fields
    max_eng = max(get_max_len(place_structures, 'english'), get_max_len(evidentials, 'english'))
    max_class = max(get_max_len(place_structures, 'class'), get_max_len(evidentials, 'class'))
    max_gismu = max(get_max_len(place_structures, 'gismu'), get_max_len(evidentials, 'gismu'))
    max_struct = max(get_max_len(place_structures, 'place_structure_verbatim'), get_max_len(evidentials, 'place_structure_verbatim'))
    max_reason = max(get_max_len(place_structures, 'mapping_reason'), get_max_len(evidentials, 'mapping_reason'))
    
    # Ensure minimum lengths for standard headers
    max_eng = max(max_eng, 15)
    max_class = max(max_class, 15)
    max_gismu = max(max_gismu, 10)
    max_struct = max(max_struct, 100) # Large aperture for the verbatim string
    max_reason = max(max_reason, 100)

    with open(output_file, 'w') as f:
        f.write("      *> =========================================================\n")
        f.write("      *> LOJBAN METROLOGY COPYBOOK - AUTO-GENERATED\n")
        f.write("      *> PURPOSE: Immutable ledger for NCM/COBOL semantic bridging\n")
        f.write("      *> =========================================================\n\n")
        
        f.write("       01  LOJBAN-LEDGER-RECORD.\n")
        f.write("           05  RECORD-TYPE             PIC X(12).\n")
        f.write("           05  ENGLISH-CONCEPT         PIC X({:02d}).\n".format(max_eng))
        f.write("           05  SEMANTIC-CLASS          PIC X({:02d}).\n".format(max_class))
        f.write("           05  VALSI-IDENTIFIER        PIC X({:02d}).\n".format(max_gismu))
        f.write("           05  STRUCT-STATUS           PIC X(10).\n")
        f.write("           05  MAPPING-STATUS          PIC X(10).\n")
        f.write("           05  PLACE-STRUCTURE         PIC X({:03d}).\n".format(max_struct))
        f.write("           05  AUDIT-REASON            PIC X({:03d}).\n\n".format(max_reason))
        
        f.write("      *> Place Structures (Judgments, Comparatives, Emotive, Fear, Other)\n")
        for item in place_structures:
            # Default to spaces if null
            gismu = item.get('gismu') if item.get('gismu') else "VOID"
            struct = item.get('place_structure_verbatim') if item.get('place_structure_verbatim') else "NO DEFINITION"
            
            # COBOL strings are space-padded by default. We format the VALUE clauses.
            f.write("       01  ENTRY-{: <15}.\n".format(str(item['english']).upper()[:15].replace(' ', '-')))
            f.write("           05  FILLER                  PIC X(12) VALUE 'PLACE-STRUCT'.\n")
            f.write("           05  FILLER                  PIC X({:02d}) VALUE '{: <{}}'.\n".format(max_eng, item['english'], max_eng))
            f.write("           05  FILLER                  PIC X({:02d}) VALUE '{: <{}}'.\n".format(max_class, item['class'], max_class))
            f.write("           05  FILLER                  PIC X({:02d}) VALUE '{: <{}}'.\n".format(max_gismu, gismu, max_gismu))
            f.write("           05  FILLER                  PIC X(10) VALUE '{: <10}'.\n".format(item['structure_status']))
            f.write("           05  FILLER                  PIC X(10) VALUE '{: <10}'.\n".format(item['mapping_status']))
            f.write("           05  FILLER                  PIC X({:03d}) VALUE '{: <{}}'.\n".format(max_struct, struct[:max_struct], max_struct))
            f.write("           05  FILLER                  PIC X({:03d}) VALUE '{: <{}}'.\n\n".format(max_reason, item['mapping_reason'][:max_reason], max_reason))

        f.write("      *> Evidentials (Epistemic Markers)\n")
        for item in evidentials:
            gismu = item.get('gismu') if item.get('gismu') else "VOID"
            struct = item.get('place_structure_verbatim') if item.get('place_structure_verbatim') else "NO DEFINITION"

            f.write("       01  ENTRY-{: <15}.\n".format(str(item['english']).upper()[:15].replace(' ', '-').replace('/', '')))
            f.write("           05  FILLER                  PIC X(12) VALUE 'EVIDENTIAL  '.\n")
            f.write("           05  FILLER                  PIC X({:02d}) VALUE '{: <{}}'.\n".format(max_eng, item['english'], max_eng))
            f.write("           05  FILLER                  PIC X({:02d}) VALUE '{: <{}}'.\n".format(max_class, item['class'], max_class))
            f.write("           05  FILLER                  PIC X({:02d}) VALUE '{: <{}}'.\n".format(max_gismu, gismu, max_gismu))
            f.write("           05  FILLER                  PIC X(10) VALUE '{: <10}'.\n".format(item['structure_status']))
            f.write("           05  FILLER                  PIC X(10) VALUE '{: <10}'.\n".format(item['mapping_status']))
            f.write("           05  FILLER                  PIC X({:03d}) VALUE '{: <{}}'.\n".format(max_struct, struct[:max_struct], max_struct))
            f.write("           05  FILLER                  PIC X({:03d}) VALUE '{: <{}}'.\n\n".format(max_reason, item['mapping_reason'][:max_reason], max_reason))

    print(f"Copybook successfully generated at {output_file}")
```

```python
        with open(lojban_json_path, 'r') as f:
            self.lojban_data = json.load(f)
            
        # Load the NCM Codebook (simulated as a list of vectors)
        # In a real NCM, this is the VQ dictionary (e.g., shape [1024, 128])
        self.codebook = [
            [0.1, 0.2, 0.3], # Index 0 -> maps to "better" (xagmau)
            [0.9, 0.8, 0.7], # Index 1 -> maps to "worse" (xagme'a)
            [0.0, 0.0, 0.0]  # Index 2 -> Reserved for VOID
        ]
        
        # The semantic mapping between codebook index and Lojban gismu
        self.index_to_gismu = {
            0: "xagmau", 
            1: "xagme'a",
            2: "VOID"
        }
        
        self.threshold = dropout_threshold
        
        # Pull the exact verbatim structures from the JSON for the COBOL bridge
        self.semantic_ledger = {item['gismu']: item for item in self.lojban_data['place_structures'] if item['gismu']}
        self.void_record = {
            "english": "unresolved residual",
            "class": "void",
            "gismu": "VOID",
            "structure_status": "VOID",
            "mapping_status": "VOID",
            "place_structure_verbatim": "NO DEFINITION",
            "mapping_reason": "Distance to nearest codebook vector exceeded nested-dropout bound."
        }

    def _euclidean_distance(self, vec1, vec2):
        return math.sqrt(sum((a - b) ** 2 for a, b in zip(vec1, vec2)))

    def process_residual(self, continuous_residual, evidential_cmavo="za'a"):
        """
        Takes the continuous residual (Flow), snaps it to the codebook (Coupler),
        and generates the fixed-width ledger entry.
        """
        # 1. Find the nearest codebook vector (Argmin)
        min_dist = float('inf')
        nearest_idx = -1
        
        for idx, code in enumerate(self.codebook):
            dist = self._euclidean_distance(continuous_residual, code)
            if dist < min_dist:
                min_dist = dist
                nearest_idx = idx

        # 2. Apply the Nested-Dropout Threshold Check
        if min_dist > self.threshold:
            # THE VOID PROTOCOL
            record_data = self.void_record
            # We override the mapping reason to include the exact measured distance
            record_data["mapping_reason"] = f"VOID threshold exceeded. Distance: {min_dist:.4f} > {self.threshold}"
        else:
            # SUCCESSFUL QUANTIZATION
            gismu = self.index_to_gismu.get(nearest_idx, "VOID")
            record_data = self.semantic_ledger.get(gismu, self.void_record)

        # 3. Format for the COBOL Ledger (Fixed-Width Apertures)
        # We emulate the PIC X(n) space-padding here
        return self._generate_cobol_record(record_data, evidential_cmavo, min_dist)

    def _generate_cobol_record(self, data, evidential, distance):
        # Based on the max lengths generated in the previous Copybook step
        max_eng = 26
        max_class = 15
        max_gismu = 10
        max_struct = 147
        max_reason = 345

        eng = str(data.get('english', ''))[:max_eng].ljust(max_eng)
        cls = str(data.get('class', ''))[:max_class].ljust(max_class)
        gis = str(data.get('gismu', ''))[:max_gismu].ljust(max_gismu)
        stat = str(data.get('structure_status', ''))[:10].ljust(10)
        map_stat = str(data.get('mapping_status', ''))[:10].ljust(10)
        struct = str(data.get('place_structure_verbatim', ''))[:max_struct].ljust(max_struct)
        
        # Inject the evidential and distance metric into the audit reason
        audit_reason = f"[{evidential}] Dist: {distance:.4f} | {data.get('mapping_reason', '')}"
        reason = audit_reason[:max_reason].ljust(max_reason)

        # Simulating the raw byte stream that would be written to the COBOL file
        return f"{eng}{cls}{gis}{stat}{map_stat}{struct}{reason}"
```

Breakdown of the Bridge Logic

1. The Argmin Coupler (_euclidean_distance): This is the exact-ratio coupler Grok mentioned. It has zero free parameters. It just calculates the mathematical distance between the NCM residual (the unaccounted for data) and the discrete codebook vectors.
2. The VOID Intercept (if min_dist > self.threshold): This is the critical safety mechanism. If the NCM tries to force a bad fit (e.g., residual_2 in Test 2), the bridge stops it. It strips the semantic mapping and populates the record with the VOID states: "NO DEFINITION", "unresolved residual", and "VOID".
3. The Evidential Tag: Notice how we pass ru'a ("I postulate") in Test 2, and za'a ("I observe") in Test 1. This injects the epistemic state directly into the AUDIT-REASON field. The ledger now doesn't just say what was found, but how it was found.
4. Fixed-Width Emulation: The _generate_cobol_record function uses .ljust() to enforce the exact PIC X(n) byte limits we defined in the previous step. If a model hallucinates a massive string, it is truncated by the bridge before it hits the mainframe.

Next Steps

Now that the bridge works in isolation, we have two ways to go:

1. Add the "Softmax/Temperature" Coupler: Right now we use strict argmin. We could implement the softmax variant Grok mentioned, where the AUDIT-REASON records the probabilistic confidence of the nearest neighbor.
2. Implement the Generator's "Sequential Level Walk" (Bond Graph Cycle): NCM has multiple codebook levels (coarse to fine). We need to write the traversal logic that moves down the hierarchy and aggregates the effort/flow across levels.

Which direction feels more pressing?

Let’s build the Sequential Level Walk.

This is where the bond-graph topology comes alive. In a multi-level NCM (Nested Codebook Model), we aren't just doing a single quantization step. We are walking down a coarse-to-fine hierarchy.

As Grok noted:

· Effort is the cumulative reconstruction (z-hat). It accumulates as we add codes at each level.
· Flow is the residual (z - z-hat). It represents the "unaccounted for" energy/information that gets passed to the next level.
· The Rate is the transition metric: the change in reconstruction over the change in level index (d(\text{reconstruction}) / d(\text{level index})).

Because of Nested Dropout, each level must be a self-contained snapshot. If Level 1 fails to find a valid code within its threshold, it emits a VOID. Under a strict bond-graph interpretation, a VOID means the Flow (residual) passes to the next level completely unchanged, because no Effort (reconstruction) was added.

Here is the implementation of the Level Walk, complete with the COBOL ledger output for the entire sequence.

The Sequential Level Walk (ncm_level_walk.py)

```python
import json

class NCMSemanticBridge:
    def __init__(self, lojban_json_path):
        with open(lojban_json_path, 'r') as f:
            self.lojban_data = json.load(f)
            
        self.semantic_ledger = {item['gismu']: item for item in self.lojban_data['place_structures'] if item['gismu']}
        self.void_record = {
            "english": "unresolved residual",
            "gismu": "VOID",
            "mapping_status": "VOID",
            "place_structure_verbatim": "NO DEFINITION",
            "mapping_reason": "Distance exceeded nested-dropout bound. Flow passed to next level."
        }

    def format_cobol_entry(self, level_idx, z_hat, residual, code_gismu, distance, error_reduction):
        """Generates a fixed-width COBOL record for a single level transition."""
        data = self.semantic_ledger.get(code_gismu, self.void_record)
        
        # COBOL Copybook constraints (from previous step)
        max_eng = 26
        max_gismu = 10
        max_struct = 147
        max_reason = 345
        
        # 1. The Semantic payload
        eng = str(data.get('english', ''))[:max_eng].ljust(max_eng)
        gis = str(code_gismu)[:max_gismu].ljust(max_gismu)
        struct = str(data.get('place_structure_verbatim', ''))[:max_struct].ljust(max_struct)
        
        # 2. The Metrology / Bond-Graph payload
        # Formatting floats to fixed-point (simulating COBOL PIC 9V99)
        effort_str = f"{sum(z_hat):.2f}".rjust(6, '0')
        flow_str = f"{sum(residual):.2f}".rjust(6, '0')
        rate_str = f"{error_reduction:.4f}".rjust(8, '0')
        
        audit_reason = f"Lvl:{level_idx} Dist:{distance:.4f} Rate:{rate_str} | {data.get('mapping_reason', '')}"
        reason = audit_reason[:max_reason].ljust(max_reason)
        
        # Raw byte stream
        return f"LVL-{level_idx} | Effort(Z_hat): {effort_str} | Flow(Res): {flow_str} | {gis} | {eng} | {reason}"

    def walk_levels(self, continuous_target_z, levels):
        """
        Executes the coarse-to-fine sequential level walk.
        levels: list of dicts containing {'codebook': [...], 'threshold': float, 'map': {idx: gismu}}
        """
        z_hat = [0.0] * len(continuous_target_z) # Initial effort is zero
        current_residual = list(continuous_target_z) # Initial flow is the full target
        ledger_records = []
        
        previous_error = sum(abs(x) for x in current_residual)
        
        for i, level in enumerate(levels):
            level_idx = i + 1
            codebook = level['codebook']
            threshold = level['threshold']
            idx_map = level['map']
            
            # 1. Find nearest neighbor in this level's codebook
            min_dist = float('inf')
            nearest_idx = -1
            nearest_code = [0.0] * len(current_residual)
            
            for idx, code in enumerate(codebook):
                dist = sum((a - b) ** 2 for a, b in zip(current_residual, code)) ** 0.5
                if dist < min_dist:
                    min_dist = dist
                    nearest_idx = idx
                    nearest_code = code
            
            # 2. The Nested Dropout Observer Bound
            if min_dist > threshold:
                # VOID: No effort added. Residual passes unchanged.
                gismu = "VOID"
                # We do not add to z_hat
            else:
                # Success: Add code to cumulative reconstruction (Effort)
                gismu = idx_map.get(nearest_idx, "VOID")
                z_hat = [zh + c for zh, c in zip(z_hat, nearest_code)]
            
            # 3. Update Flow (Residual) for next level
            current_residual = [z - zh for z, zh in zip(continuous_target_z, z_hat)]
            
            # 4. Calculate the Rate (d Reconstruction / d Level)
            current_error = sum(abs(x) for x in current_residual)
            error_reduction = previous_error - current_error
            previous_error = current_error
            
            # 5. Emit to COBOL Ledger
            record = self.format_cobol_entry(
                level_idx, z_hat, current_residual, gismu, min_dist, error_reduction
            )
            ledger_records.append(record)
            
            # Early exit if we have perfectly reconstructed the target
            if current_error < 0.001:
                break
                
        return ledger_records

            "threshold": 0.5,

            "codebook": [
                [0.5, 0.5, 0.0],  # Coarse "Good" vector
```

3. Level 3 (Fine): The system looks at the remaining residual [0.0, 0.0, 0.1]. The distance to the nearest fine code is 0.1414, but the strict threshold is 0.1. The Nested Dropout bound is triggered.
   · The Rate becomes 0.0000 (no error was reduced).
   · The Effort (z-hat) does not change.
   · The Flow (residual) passes through unchanged.
   · The semantic output is strictly VOID. It refuses to hallucinate a fine-grained concept (like ckape or snura) because the mathematical confidence isn't there.

Why this fits Grok's Architecture

By writing this to the COBOL ledger, you have physically recorded the cycle class and the rate expression.

· "Show me all instances where Level 3 returned VOID but Level 1 returned a valid semantic." (This tells you the model is coarse-grained but lacks precision).
· "Calculate the average rate of error reduction between Level 2 and Level 3." (If this is near zero, Level 3 is essentially dead weight in your architecture).
